function fmt(n) {
  if (Math.abs(n) >= 1e9) return '$' + (n/1e9).toFixed(1) + 'B';
  if (Math.abs(n) >= 1e6) return '$' + (n/1e6).toFixed(1) + 'M';
  if (Math.abs(n) >= 1e3) return '$' + Math.round(n/1000) + 'K';
  return '$' + Math.round(n).toLocaleString();
}
function calc_ci() {
  const spend = +document.getElementById('ci_spend').value;
  const win = +document.getElementById('ci_win').value / 100;
  const deal = +document.getElementById('ci_deal').value;

  document.getElementById('ci_spend_out').textContent = fmt(spend);
  document.getElementById('ci_win_out').textContent = Math.round(win*100) + '%';
  document.getElementById('ci_deal_out').textContent = fmt(deal);

  const dealsPerMonth = spend / 2000; // Proxy for volume
  const currentRev = dealsPerMonth * win * deal;
  const lift = 0.24; // 24% relative win rate lift
  const newWin = win * (1 + lift);
  const newRev = dealsPerMonth * newWin * deal;
  const addRev = newRev - currentRev;

  document.getElementById('ci_new_win').textContent = (newWin*100).toFixed(1) + '%';
  document.getElementById('ci_rev').textContent = fmt(addRev);
  document.getElementById('ci_annual').textContent = fmt(addRev * 12);
}
calc_ci();
</script>
